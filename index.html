LOGIN SISWA & GURU if (action === "login") { const usernameInput = String((postData && postData.username) || params.username || "").trim().toLowerCase(); const pinInput = String((postData && postData.pin) || params.pin || "").trim(); const sheetUsers = ss.getSheets().find(s => s.getName().trim().toLowerCase() === "users"); if (!sheetUsers) { return kirimJSON({ status: "error", message: "Tab 'Users' tidak ditemukan!" }); } const rows = sheetUsers.getDataRange().getValues(); const headers = rows[0].map(h => String(h).trim().toLowerCase()); const idxUser = headers.indexOf("username"); const idxPin = headers.indexOf("pin"); const idxId = headers.indexOf("user_id"); const idxNama = headers.indexOf("nama_lengkap"); const idxRole = headers.indexOf("role"); const idxKelas = headers.indexOf("kelas"); for (let i = 1; i < rows.length; i++) { const uSheet = String(rows[i][idxUser] || "").trim().toLowerCase(); const pSheet = String(rows[i][idxPin] || "").trim(); if (uSheet === usernameInput && pSheet === pinInput) { return kirimJSON({ status: "success", user: { user_id: String(rows[i][idxId] || "").trim(), nama_lengkap: String(rows[i][idxNama] || "").trim(), username: String(rows[i][idxUser] || "").trim(), role: String(rows[i][idxRole] || "siswa").trim().toLowerCase(), Kelas: String(rows[i][idxKelas] || "").trim() } }); } } return kirimJSON({ status: "error", message: "Username atau PIN tidak cocok. Pastikan memakai NIS 7 digit!" }); } // 2. AMBIL SOAL & MATERI if (action === "getSoal" || action === "getMateri") { return kirimJSON({ status: "success", soal: ambilDataSheet(ss, "Bank_Soal"), materi: ambilDataSheet(ss, "Materi"), kursus: ambilDataSheet(ss, "Kursus") }); } // 3. SIMPAN HASIL KUIS OTOMATIS KE TAB Nilai_Tugas if (action === "simpanNilai") { const sheetNilai = pastikanSheet(ss, "Nilai_Tugas", [ "timestamp", "user_id", "nama_lengkap", "materi_id", "jenis_aktivitas", "skor_nilai", "link_file_drive" ]); const waktuSekarang = Utilities.formatDate(new Date(), "Asia/Makassar", "dd/MM/yyyy HH:mm:ss"); const userId = String((postData && postData.user_id) || params.user_id || "").trim(); const namaLengkap = String((postData && postData.nama_lengkap) || params.nama_lengkap || "").trim(); const materiId = String((postData && postData.materi_id) || params.materi_id || "").trim(); const skorNilai = Number((postData && postData.skor_nilai) || params.skor_nilai || 0); sheetNilai.appendRow([waktuSekarang, userId, namaLengkap, materiId, "Kuis Otomatis", skorNilai, "-"]); return kirimJSON({ status: "success", message: "Nilai berhasil direkam ke Nilai_Tugas!" }); } // 4. REKAP NILAI LIVE UNTUK RUANG KENDALI GURU if (action === "getRekapGuru") { const dataNilai = ambilDataSheet(ss, "Nilai_Tugas"); const dataUsers = ambilDataSheet(ss, "Users"); const mapKelas = {}; dataUsers.forEach(u => { mapKelas[String(u.user_id || "").trim()] = String(u.Kelas || u.kelas || "-").trim(); }); const gabungan = dataNilai.map(row => ({ timestamp: row.timestamp || "-", user_id: row.user_id || "-", nama_lengkap: row.nama_lengkap || "-", kelas: mapKelas[String(row.user_id || "").trim()] || "-", materi_id: row.materi_id || "-", jenis_aktivitas: row.jenis_aktivitas || "-", skor_nilai: row.skor_nilai || 0 })).reverse(); return kirimJSON({ status: "success", rekap: gabungan }); } return kirimJSON({ status: "error", message: "Aksi tidak dikenali: " + action }); } catch (err) { return kirimJSON({ status: "error", message: "Error Server: " + err.toString() }); } } function ambilDataSheet(ss, namaTab) { const sheet = ss.getSheets().find(s => s.getName().trim().toLowerCase() === namaTab.toLowerCase()); if (!sheet) return []; const rows = sheet.getDataRange().getValues(); if (rows.length < 2) return []; const headers = rows[0].map(h => String(h).trim()); const hasil = []; for (let i = 1; i < rows.length; i++) { if (rows[i].join("").trim() === "") continue; const obj = {}; headers.forEach((h, idx) => { obj[h] = rows[i][idx]; }); hasil.push(obj); } return hasil; } function pastikanSheet(ss, namaTab, headerDefault) { let sheet = ss.getSheets().find(s => s.getName().trim().toLowerCase() === namaTab.toLowerCase()); if (!sheet) { sheet = ss.insertSheet(namaTab); sheet.appendRow(headerDefault); } return sheet; } function kirimJSON(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } Portal Belajar Digital — UPT SMA Negeri 26 Bone
